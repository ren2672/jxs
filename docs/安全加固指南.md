## 安全风险评估

### 风险等级定义
- 🔴 **严重**: 可导致系统完全被攻破或数据泄露
- 🟠 **高**: 可导致权限提升或数据篡改
- 🟡 **中**: 可导致信息泄露或服务中断
- 🟢 **低**: 轻微安全问题，建议修复

```python
import bcrypt
import hashlib

# 新的密码哈希函数
def hash_password(password: str) -> str:
    """使用bcrypt哈希密码"""
    salt = bcrypt.gensalt(rounds=12)
    return bcrypt.hashpw(password.encode('utf-8'), salt).decode('utf-8')

def verify_password(password: str, hashed: str) -> bool:
    """验证密码"""
    return bcrypt.checkpw(password.encode('utf-8'), hashed.encode('utf-8'))

# 修改登录验证逻辑
```

### 2. 修复SQL注入（严重）

#### 2.1 修复所有动态SQL

```python
# 错误的做法（存在SQL注入）
cursor.execute(f"SELECT * FROM products WHERE name LIKE '%{search_term}%'")

# 正确的做法（使用参数化查询）
cursor.execute("SELECT * FROM products WHERE name LIKE ?", (f'%{search_term}%',))

# 修复动态WHERE子句构建
def build_safe_query(base_query: str, filters: dict) -> tuple:
    """安全构建动态查询"""
    allowed_fields = {
        'name': 'name LIKE ?',
        'category': 'category = ?',
        'warehouse': 'warehouse = ?'
    }

    where_clauses = []
    params = []

    for field, value in filters.items():
        if field in allowed_fields and value:
            where_clauses.append(allowed_fields[field])
            if 'LIKE' in allowed_fields[field]:
                params.append(f'%{value}%')
            else:
                params.append(value)

    if where_clauses:
        query = f"{base_query} WHERE {' AND '.join(where_clauses)}"
        return query, params
    return base_query, []
```

```python
class PermissionManager:
    """权限管理器"""

    PERMISSIONS = {
        'admin': [
            'user.manage', 'product.manage', 'inventory.manage',
            'report.view', 'report.export', 'system.config'
        ],
        'manager': [
            'product.view', 'inventory.view', 'inventory.in',
            'inventory.out', 'report.view', 'report.export'
        ],
        'operator': [
            'product.view', 'inventory.view', 'inventory.in',
            'inventory.out', 'report.view'
        ]
    }

    @classmethod
    def has_permission(cls, username: str, permission: str) -> bool:
        """检查用户权限"""
        conn = sqlite3.connect(DB_FILE)
        cursor = conn.cursor()

        cursor.execute("""
            SELECT u.role, p.permissions
            FROM users u
            LEFT JOIN user_permissions p ON u.id = p.user_id
            WHERE u.username = ? AND u.active = 1
        """, (username,))

        result = cursor.fetchone()
        conn.close()

        if not result:
            return False

        role, extra_perms = result
        role_perms = cls.PERMISSIONS.get(role, [])
        all_perms = role_perms + (extra_perms.split(',') if extra_perms else [])

        return permission in all_perms

    @classmethod
    def require_permission(cls, permission: str):
        """权限检查装饰器"""
        def decorator(func):
            def wrapper(self, *args, **kwargs):
                if not hasattr(self, 'current_user'):
                    raise PermissionError("未登录")
                if not cls.has_permission(self.current_user, permission):
                    raise PermissionError(f"权限不足，需要 {permission} 权限")
                return func(self, *args, **kwargs)
            return wrapper
        return decorator
```

## 密码安全加固

### 4.1 强化密码策略

```python
class PasswordPolicy:
    """密码策略配置"""

    MIN_LENGTH = 8
    MAX_LENGTH = 32
    REQUIRE_UPPER = True
    REQUIRE_LOWER = True
    REQUIRE_NUMBER = True
    REQUIRE_SPECIAL = True
    FORBIDDEN_PATTERNS = [
        '123456', 'password', 'admin', 'qwerty',
        '111111', '000000', '123123'
    ]

    @classmethod
    def validate(cls, password: str, username: str = None) -> tuple:
        """验证密码强度"""
        errors = []

        # 长度检查
        if len(password) < cls.MIN_LENGTH:
            errors.append(f"密码长度不能少于{cls.MIN_LENGTH}位")
        if len(password) > cls.MAX_LENGTH:
            errors.append(f"密码长度不能超过{cls.MAX_LENGTH}位")

        # 复杂度检查
        if cls.REQUIRE_UPPER and not any(c.isupper() for c in password):
            errors.append("密码必须包含大写字母")
        if cls.REQUIRE_LOWER and not any(c.islower() for c in password):
            errors.append("密码必须包含小写字母")
        if cls.REQUIRE_NUMBER and not any(c.isdigit() for c in password):
            errors.append("密码必须包含数字")
        if cls.REQUIRE_SPECIAL and not any(c in '!@#$%^&*()_+-=[]{}|;:,.<>?' for c in password):
            errors.append("密码必须包含特殊字符")

        # 弱密码检查
        if password.lower() in cls.FORBIDDEN_PATTERNS:
            errors.append("不能使用常见弱密码")

        # 用户名检查
        if username and username.lower() in password.lower():
            errors.append("密码不能包含用户名")

        return len(errors) == 0, errors
```

```python
def check_password_expiry(username: str) -> tuple:
    """检查密码是否过期"""
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()

    cursor.execute("""
        SELECT password_changed_date
        FROM users
        WHERE username = ?
    """, (username,))

    result = cursor.fetchone()
    conn.close()

    if not result:
        return False, "用户不存在"

    changed_date = datetime.fromisoformat(result[0])
    days_since_change = (datetime.now() - changed_date).days

    if days_since_change > 90:
        return False, "密码已过期，请重置密码"
    elif days_since_change > 80:
        return True, "密码将在10天内过期，建议尽快更改"

    return True, "密码正常"
```

```python
class SecureDatabase:
    """安全数据库访问类"""

    def __init__(self):
        self.conn = None
        self.user = None
        self.permissions = []

    def connect(self, username: str, password: str):
        """安全连接数据库"""
        # 验证用户身份
        if not verify_login(username, password):
            raise SecurityError("认证失败")

        # 加载用户权限
        self.user = username
        self.permissions = get_user_permissions(username)

        # 建立连接
        self.conn = sqlite3.connect(DB_FILE)
        self.conn.execute("PRAGMA foreign_keys = ON")
        self.conn.row_factory = sqlite3.Row

    def execute(self, query: str, params: tuple = ()) -> sqlite3.Cursor:
        """安全执行查询"""
        # 验证查询权限
        if not self._check_query_permission(query):
            raise PermissionError("无权限执行此操作")

        # 记录查询日志
        self._log_query(query, params)

        return self.conn.execute(query, params)

    def _check_query_permission(self, query: str) -> bool:
        """检查查询权限"""
        query_lower = query.lower().strip()

        # 检查操作类型
        if query_lower.startswith('select'):
            required_perm = 'data.read'
        elif query_lower.startswith(('insert', 'update', 'delete')):
            required_perm = 'data.write'
        elif query_lower.startswith(('create', 'alter', 'drop')):
            required_perm = 'schema.modify'
        else:
            return False

        return required_perm in self.permissions
```

### 6.3 CSRF防护

```python
import secrets
import time

class CSRFProtection:
    """CSRF防护"""

    @staticmethod
    def generate_token() -> str:
        """生成CSRF令牌"""
        return secrets.token_urlsafe(32)

    @staticmethod
    def verify_token(token: str, session_token: str, max_age: int = 3600) -> bool:
        """验证CSRF令牌"""
        if not token or not session_token:
            return False

        # 解析令牌中的时间戳
        try:
            token_data = token.split(':')
            if len(token_data) != 2:
                return False

            token_value, timestamp = token_data
            age = int(time.time()) - int(timestamp)

            if age > max_age:
                return False

            return secrets.compare_digest(token_value, session_token)
        except:
            return False
```

```python
import requests
from requests.packages.urllib3.exceptions import InsecureRequestWarning

class SecureSync:
    """安全同步客户端"""

    def __init__(self, api_url: str, api_key: str):
        self.api_url = api_url
        self.api_key = api_key
        self.session = requests.Session()
        # 禁用不安全请求警告（仅测试环境）
        requests.packages.urllib3.disable_warnings(InsecureRequestWarning)

    def sync_data(self, data: dict) -> bool:
        """安全同步数据"""
        try:
            # 加密数据
            encrypted_data = self._encrypt_data(data)

            # 发送请求
            response = self.session.post(
                f"{self.api_url}/sync",
                json={
                    'api_key': self.api_key,
                    'data': encrypted_data,
                    'timestamp': int(time.time())
                },
                verify=True,  # 生产环境必须验证SSL
                timeout=30
            )

            return response.status_code == 200
        except Exception as e:
            log_error(f"同步失败: {str(e)}")
```

### 9.1 实时监控

```python
import threading
import queue
from datetime import datetime

class SecurityMonitor:
    """安全监控器"""

    def __init__(self):
        self.alert_queue = queue.Queue()
        self.monitoring = False

    def start_monitoring(self):
        """启动监控"""
        self.monitoring = True
        threading.Thread(target=self._monitor_loop, daemon=True).start()

    def _monitor_loop(self):
        """监控循环"""
        while self.monitoring:
            # 检查实时威胁
            self._check_real_time_threats()

            # 处理告警
            while not self.alert_queue.empty():
                alert = self.alert_queue.get()
                self._handle_alert(alert)

            time.sleep(5)  # 5秒检查一次

    def _check_real_time_threats(self):
        """检查实时威胁"""
        # 检查暴力破解
        failed_logins = get_recent_failed_logins(minutes=5)
        if len(failed_logins) > 5:
            self.alert_queue.put({
                'type': 'BRUTE_FORCE',
                'severity': 'HIGH',
                'details': f"5分钟内{len(failed_logins)}次失败登录"
            })

        # 检查异常数据访问
        unusual_access = detect_unusual_access()
        for access in unusual_access:
            self.alert_queue.put({
                'type': 'UNUSUAL_ACCESS',
                'severity': 'MEDIUM',
                'details': access
            })

    def _handle_alert(self, alert: dict):
        """处理告警"""
        # 记录日志
```
