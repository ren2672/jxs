## 应急响应流程

### 1.1 事件分级

| 级别 | 定义 | 响应时间 | 影响范围 |
|------|------|----------|----------|
| **P0 - 紧急** | 系统完全不可用，核心业务中断 | 15分钟内 | 全公司 |
| **P1 - 高** | 系统严重异常，核心功能受影响 | 30分钟内 | 多部门 |
| **P2 - 中** | 系统部分功能异常，影响个别部门 | 2小时内 | 单个部门 |
| **P3 - 低** | 非核心功能问题，对业务影响较小 | 4小时内 | 少数用户 |

### 1.2 应急响应流程图

```
事件发生 → 初步评估 → 启动响应 → 事件处理 → 恢复验证 → 事后总结
    ↓          ↓          ↓          ↓          ↓          ↓
  发现问题  → 确定级别  → 分配任务  → 执行方案  → 测试确认  → 改进措施
    ↓          ↓          ↓          ↓          ↓          ↓
  通知人员  → 上报领导  → 资源调配  → 进度跟踪  → 用户通知  → 文档更新
```

#### 2.1.2 紧急恢复流程

```python
# emergency_recovery.py
import os
import shutil
import sqlite3
import time
from datetime import datetime

class EmergencyRecovery:
    """紧急恢复处理"""

    def __init__(self):
        self.backup_dir = "C:\\InventoryPro\\data\\backup"
        self.data_dir = "C:\\InventoryPro\\data"
        self.log_file = os.path.join(self.data_dir, "logs", f"emergency_recovery_{datetime.now().strftime('%Y%m%d_%H%M%S')}.log")

    def log(self, message):
        """记录日志"""
        timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
        log_entry = f"{timestamp} - {message}"
        print(log_entry)
        with open(self.log_file, 'a', encoding='utf-8') as f:
            f.write(log_entry + '\n')

    def emergency_restore(self):
        """紧急恢复主流程"""
        self.log("开始紧急恢复流程")

        try:
            # 1. 停止所有服务
            self.stop_services()

            # 2. 诊断问题
            issue_type = self.diagnose_issue()

            # 3. 执行恢复
            if issue_type == "database_corrupted":
                self.restore_database()
            elif issue_type == "service_crashed":
                self.restart_services()
            elif issue_type == "disk_full":
                self.cleanup_disk()
            else:
                self.full_system_restore()

            # 4. 验证恢复
            if self.verify_recovery():
                self.log("恢复成功")
                return True
            else:
                self.log("恢复失败，执行完整恢复")
                self.full_system_restore()
                return self.verify_recovery()

        except Exception as e:
            self.log(f"恢复过程出错: {str(e)}")
            return False

    def diagnose_issue(self):
        """诊断问题类型"""
        self.log("开始诊断问题...")

        # 检查数据库完整性
        db_path = os.path.join(self.data_dir, "inventory.db")
        if os.path.exists(db_path):
            try:
                conn = sqlite3.connect(db_path)
                cursor = conn.cursor()
                cursor.execute("PRAGMA integrity_check")
                result = cursor.fetchone()
                conn.close()

                if result[0] != "ok":
                    self.log("检测到数据库损坏")
                    return "database_corrupted"
            except Exception as e:
                self.log(f"数据库访问失败: {str(e)}")
                return "database_corrupted"

        # 检查磁盘空间
        disk_usage = shutil.disk_usage("C:\\")
        free_percent = (disk_usage.free / disk_usage.total) * 100
        if free_percent < 5:
            self.log(f"磁盘空间不足: {free_percent:.2f}%")
            return "disk_full"

        # 检查服务状态
        if not self.is_service_running():
            self.log("服务未运行")
            return "service_crashed"

        self.log("未检测到明确问题")
        return "unknown"

    def restore_database(self):
        """恢复数据库"""
        self.log("开始恢复数据库...")

        # 查找最新备份
        latest_backup = self.find_latest_backup()
        if not latest_backup:
            raise Exception("找不到有效的数据库备份")

        self.log(f"使用备份: {latest_backup}")

        # 备份当前损坏的数据库
        corrupted_db = os.path.join(self.data_dir, "inventory.db")
        if os.path.exists(corrupted_db):
            corrupted_backup = os.path.join(
                self.backup_dir,
                f"corrupted_{datetime.now().strftime('%Y%m%d_%H%M%S')}.db"
            )
            shutil.move(corrupted_db, corrupted_backup)
            self.log(f"已备份损坏的数据库到: {corrupted_backup}")

        # 恢复数据库
        shutil.copy2(latest_backup, corrupted_db)
        self.log("数据库恢复完成")

    def find_latest_backup(self):
        """查找最新的有效备份"""
        import glob

        backup_pattern = os.path.join(self.backup_dir, "inventory_*.db")
        backup_files = glob.glob(backup_pattern)

        # 按修改时间排序，获取最新的
        backup_files.sort(key=lambda x: os.path.getmtime(x), reverse=True)

        for backup_file in backup_files:
            # 验证备份完整性
            try:
                conn = sqlite3.connect(backup_file)
                cursor = conn.cursor()
                cursor.execute("PRAGMA integrity_check")
                result = cursor.fetchone()
                conn.close()

                if result[0] == "ok":
                    return backup_file
            except:
                continue

        return None

    def full_system_restore(self):
        """完整系统恢复"""
        self.log("开始完整系统恢复...")

        # 1. 恢复数据库
        self.restore_database()

        # 2. 恢复配置文件
        config_backup = os.path.join(self.backup_dir, "settings.json.bak")
        if os.path.exists(config_backup):
            shutil.copy2(config_backup, os.path.join(self.data_dir, "settings.json"))
            self.log("配置文件已恢复")

        # 3. 清理临时文件
        temp_dir = os.path.join(self.data_dir, "temp")
        if os.path.exists(temp_dir):
            shutil.rmtree(temp_dir)
            os.makedirs(temp_dir)
            self.log("临时文件已清理")

        # 4. 重新启动服务
        self.restart_services()

    def stop_services(self):
        """停止所有服务"""
        self.log("停止服务...")
        os.system("taskkill /F /IM InventoryPro.exe 2>nul")
        time.sleep(3)

    def restart_services(self):
        """重启服务"""
        self.log("重启服务...")
        exe_path = "C:\\InventoryPro\\InventoryPro.exe"
        if os.path.exists(exe_path):
            os.startfile(exe_path)
            time.sleep(5)

    def is_service_running(self):
        """检查服务是否运行"""
        import psutil
        for proc in psutil.process_iter(['name']):
            if proc.info['name'] == 'InventoryPro.exe':
                return True
        return False

    def verify_recovery(self):
        """验证恢复结果"""
        self.log("验证恢复结果...")

        # 检查进程
        if not self.is_service_running():
            self.log("服务未启动")
            return False

        # 检查数据库
        try:
            conn = sqlite3.connect(os.path.join(self.data_dir, "inventory.db"))
            cursor = conn.cursor()
            cursor.execute("SELECT COUNT(*) FROM products")
            count = cursor.fetchone()[0]
            conn.close()

            if count > 0:
                self.log(f"验证成功: 找到 {count} 个商品")
                return True
            else:
                self.log("数据库为空")
                return False
        except Exception as e:
            self.log(f"数据库验证失败: {str(e)}")
            return False

    def cleanup_disk(self):
        """清理磁盘空间"""
        self.log("清理磁盘空间...")

        # 清理日志文件
        import glob
        log_dir = os.path.join(self.data_dir, "logs")
        for log_file in glob.glob(os.path.join(log_dir, "*.log")):
            if os.path.getsize(log_file) > 100 * 1024 * 1024:  # 大于100MB
                os.remove(log_file)
                self.log(f"删除大日志文件: {log_file}")

        # 清理临时文件
        temp_dir = os.path.join(self.data_dir, "temp")
        if os.path.exists(temp_dir):
            shutil.rmtree(temp_dir)
            os.makedirs(temp_dir)

        # 清理旧备份（保留最近5个）
        backup_pattern = os.path.join(self.backup_dir, "inventory_*.db")
        backup_files = glob.glob(backup_pattern)
        backup_files.sort(key=lambda x: os.path.getmtime(x), reverse=True)

        for old_backup in backup_files[5:]:
            os.remove(old_backup)
            self.log(f"删除旧备份: {old_backup}")

        self.log("磁盘清理完成")

# 执行紧急恢复
if __name__ == "__main__":
    recovery = EmergencyRecovery()

    # 发送紧急通知
    EmergencyContactSystem.send_emergency_alert(
        'P0',
        '进销存系统紧急恢复启动',
        '系统检测到严重故障，正在执行紧急恢复流程'
    )

    # 执行恢复
    if recovery.emergency_restore():
        # 发送恢复成功通知
        EmergencyContactSystem.send_emergency_alert(
            'P1',
            '系统恢复成功',
            '进销存系统已恢复正常运行'
        )
    else:
        # 发送恢复失败通知
        EmergencyContactSystem.send_emergency_alert(
            'P0',
            '系统恢复失败',
            '需要人工介入处理'
        )
```

```python
# database_repair.py
import sqlite3
import shutil
import os

class DatabaseRepair:
    """数据库修复工具"""

    def __init__(self):
        self.db_path = "C:\\InventoryPro\\data\\inventory.db"
        self.backup_path = "C:\\InventoryPro\\data\\backup"

    def repair_database(self):
        """修复数据库"""
        print("开始数据库修复...")

        # 1. 尝试使用SQLite修复
        if self.try_sqlite_repair():
            print("SQLite修复成功")
            return True

        # 2. 尝试从WAL文件恢复
        if self.try_wal_recovery():
            print("WAL恢复成功")
            return True

        # 3. 尝试导出数据重建
        if self.try_export_rebuild():
            print("数据重建成功")
            return True

        print("所有修复方法都失败了")
        return False

    def try_sqlite_repair(self):
        """尝试SQLite内置修复"""
        try:
            # 创建临时数据库
            temp_db = self.db_path + ".repair"
            shutil.copy2(self.db_path, temp_db)

            # 连接并尝试修复
            conn = sqlite3.connect(temp_db)
            cursor = conn.cursor()

            # 尝试快速检查
            cursor.execute("PRAGMA quick_check")
            result = cursor.fetchone()

            if result[0] == "ok":
                # 快速检查通过，尝试完整检查
                cursor.execute("PRAGMA integrity_check")
                result = cursor.fetchone()

                if result[0] == "ok":
                    conn.close()
                    shutil.move(temp_db, self.db_path)
                    return True

            # 尝试修复模式
            cursor.execute("PRAGMA writable_schema = ON")
            cursor.execute("UPDATE sqlite_master SET sql = REPLACE(sql, 'WITHOUT ROWID', '') WHERE name = 'products'")
            cursor.execute("PRAGMA writable_schema = OFF")

            conn.commit()
            conn.close()

            # 验证修复结果
            if self.verify_database(temp_db):
                shutil.move(temp_db, self.db_path)
                return True

        except Exception as e:
            print(f"SQLite修复失败: {str(e)}")

        return False

    def try_wal_recovery(self):
        """尝试从WAL文件恢复"""
        wal_file = self.db_path + "-wal"
        if not os.path.exists(wal_file):
            return False

        try:
            # 备份原文件
            shutil.copy2(self.db_path, self.db_path + ".bak")
            shutil.copy2(wal_file, wal_file + ".bak")

            # 强制检查点
            conn = sqlite3.connect(self.db_path)
            conn.execute("PRAGMA wal_checkpoint(FULL)")
            conn.close()

            # 删除WAL文件
            if os.path.exists(wal_file):
                os.remove(wal_file)

            # 验证恢复
            return self.verify_database(self.db_path)

        except Exception as e:
            print(f"WAL恢复失败: {str(e)}")
            return False

    def try_export_rebuild(self):
        """尝试导出数据重建数据库"""
        try:
            # 创建新数据库
            new_db = self.db_path + ".new"
            conn_new = sqlite3.connect(new_db)
            cursor_new = conn_new.cursor()

            # 从损坏的数据库导出数据
            conn_old = sqlite3.connect(self.db_path)
            conn_old.execute("PRAGMA journal_mode = OFF")
            conn_old.execute("PRAGMA synchronous = OFF")

            # 获取所有表
            tables = conn_old.execute(
                "SELECT name FROM sqlite_master WHERE type='table'"
            ).fetchall()

            for table in tables:
                table_name = table[0]

                # 获取表结构
                create_sql = conn_old.execute(
                    f"SELECT sql FROM sqlite_master WHERE name='{table_name}'"
                ).fetchone()

                if create_sql and create_sql[0]:
                    # 创建表
                    cursor_new.execute(create_sql[0])

                    # 尝试导出数据
                    try:
                        rows = conn_old.execute(f"SELECT * FROM {table_name}").fetchall()
                        if rows:
                            placeholders = ','.join(['?'] * len(rows[0]))
                            cursor_new.executemany(
                                f"INSERT INTO {table_name} VALUES ({placeholders})",
                                rows
                            )
                            print(f"导出表 {table_name}: {len(rows)} 行")
                    except Exception as e:
                        print(f"导出表 {table_name} 失败: {str(e)}")

            conn_new.commit()
            conn_old.close()
            conn_new.close()

            # 验证新数据库
            if self.verify_database(new_db):
                # 替换原数据库
                shutil.move(self.db_path, self.db_path + ".corrupted")
                shutil.move(new_db, self.db_path)
                return True

        except Exception as e:
            print(f"导出重建失败: {str(e)}")

        return False

    def verify_database(self, db_path):
        """验证数据库完整性"""
        try:
            conn = sqlite3.connect(db_path)
            cursor = conn.cursor()
            cursor.execute("PRAGMA integrity_check")
            result = cursor.fetchone()
            conn.close()

            return result[0] == "ok"
        except:
            return False
```

```python
        for root, dirs, files in os.walk(self.data_dir):

                    )
                    shutil.move(file_path, quarantine_path)

    def is_file_encrypted(self, file_path):
        """检查文件是否被加密"""
        try:
            # 读取文件前1KB
            with open(file_path, 'rb') as f:
                data = f.read(1024)

            # 检查熵值（加密文件通常有高熵值）
            if len(data) > 0:
                entropy = self.calculate_entropy(data)
                if entropy > 7.5:  # 高熵值阈值
                    return True

            # 检查勒索病毒标识
            ransomware_signatures = [
                b'Your files have been encrypted',
                b'All your files have been encrypted',
                b'Your personal files are encrypted'
            ]

            for signature in ransomware_signatures:
                if signature in data:
                    return True

        except:
            pass

        return False

    def calculate_entropy(self, data):
        """计算数据的熵值"""
        if len(data) == 0:
            return 0

        # 统计字节频率
        byte_counts = [0] * 256
        for byte in data:
            byte_counts[byte] += 1

        # 计算熵
        entropy = 0
        data_len = len(data)
        for count in byte_counts:
            if count > 0:
                freq = count / data_len
                entropy -= freq * (freq.bit_length() - 1)

        return entropy

            report['backup_date'] = os.path.getmtime(latest_backup)

        return report

        backup_files.sort(key=lambda x: os.path.getmtime(x), reverse=True)

        for backup in backup_files:

            if not self.is_backup_encrypted(backup):
                return backup

        return None

    def is_backup_encrypted(self, backup_path):

        try:

            cursor = conn.cursor()

            result = cursor.fetchone()
            conn.close()

            return result is None
        except:
            return True

        # 1. 完全清除感染
        self.disinfect_system()

        # 2. 从备份恢复
        self.restore_from_backup()

        # 运行杀毒软件扫描

            )

            # 恢复备份
            shutil.copy2(clean_backup, os.path.join(self.data_dir, "inventory.db"))
```
