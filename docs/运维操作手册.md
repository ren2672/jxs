```python
# monitor.py
import psutil
import sqlite3
import time
from datetime import datetime

class SystemMonitor:
    """系统监控器"""

    def __init__(self):
        self.db_path = "C:\\InventoryPro\\data\\monitoring.db"
        self.init_monitoring_db()

    def init_monitoring_db(self):
        """初始化监控数据库"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS system_metrics (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
                cpu_percent REAL,
                memory_percent REAL,
                disk_usage REAL,
                active_users INTEGER,
                db_connections INTEGER,
                response_time REAL
            )
        """)

        conn.commit()
        conn.close()

    def collect_metrics(self):
        """收集系统指标"""
        metrics = {
            'timestamp': datetime.now(),
            'cpu_percent': psutil.cpu_percent(interval=1),
            'memory_percent': psutil.virtual_memory().percent,
            'disk_usage': psutil.disk_usage('C:\\').percent,
            'active_users': self.get_active_user_count(),
            'db_connections': self.get_db_connection_count(),
            'response_time': self.measure_response_time()
        }

        # 保存到数据库
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        cursor.execute("""
            INSERT INTO system_metrics (
                cpu_percent, memory_percent, disk_usage,
                active_users, db_connections, response_time
            ) VALUES (?, ?, ?, ?, ?, ?)
        """, (
            metrics['cpu_percent'],
            metrics['memory_percent'],
            metrics['disk_usage'],
            metrics['active_users'],
            metrics['db_connections'],
            metrics['response_time']
        ))

        conn.commit()
        conn.close()

        # 检查告警
        self.check_alerts(metrics)

        return metrics

    def check_alerts(self, metrics):
        """检查告警条件"""
        alerts = []

        # CPU告警
        if metrics['cpu_percent'] > 80:
            alerts.append({
                'type': 'CPU_HIGH',
                'message': f"CPU使用率过高: {metrics['cpu_percent']}%",
                'severity': 'HIGH' if metrics['cpu_percent'] > 90 else 'MEDIUM'
            })

        # 内存告警
        if metrics['memory_percent'] > 85:
            alerts.append({
                'type': 'MEMORY_HIGH',
                'message': f"内存使用率过高: {metrics['memory_percent']}%",
                'severity': 'HIGH' if metrics['memory_percent'] > 95 else 'MEDIUM'
            })

        # 磁盘告警
        if metrics['disk_usage'] > 90:
            alerts.append({
                'type': 'DISK_FULL',
                'message': f"磁盘使用率过高: {metrics['disk_usage']}%",
                'severity': 'CRITICAL'
            })

        # 响应时间告警
        if metrics['response_time'] > 5000:  # 5秒
            alerts.append({
                'type': 'RESPONSE_SLOW',
                'message': f"响应时间过长: {metrics['response_time']}ms",
                'severity': 'MEDIUM'
            })

        # 发送告警
        for alert in alerts:
            self.send_alert(alert)

    def send_alert(self, alert):
        """发送告警"""
        # 记录告警日志
        with open("C:\\InventoryPro\\data\\logs\\alerts.log", "a") as f:
            f.write(f"{datetime.now()} - {alert['type']} - {alert['message']}\n")

        # 发送邮件（如果配置了）
        if alert['severity'] in ['HIGH', 'CRITICAL']:
            send_alert_email(alert)

# 启动监控
if __name__ == "__main__":
    monitor = SystemMonitor()

    while True:
        try:
            metrics = monitor.collect_metrics()
            print(f"监控数据收集完成: {metrics}")
            time.sleep(60)  # 每分钟收集一次
        except KeyboardInterrupt:
            print("监控停止")
            break
        except Exception as e:
            print(f"监控错误: {str(e)}")
            time.sleep(60)
```

```python
import sqlite3
from datetime import datetime, timedelta

class DataCleanup:
    """数据清理工具"""

    def __init__(self):
        self.db_path = "C:\\InventoryPro\\data\\inventory.db"

    def cleanup_old_logs(self, days=30):
        """清理旧日志"""
        cutoff_date = datetime.now() - timedelta(days=days)

        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        # 清理登录日志
        cursor.execute("""
            DELETE FROM login_logs
            WHERE login_date < ?
        """, (cutoff_date.isoformat(),))

        login_deleted = cursor.rowcount

        # 清理操作日志
        cursor.execute("""
            DELETE FROM operation_logs
            WHERE operation_date < ?
        """, (cutoff_date.isoformat(),))

        operation_deleted = cursor.rowcount

        conn.commit()
        conn.close()

        print(f"清理完成: 登录日志 {login_deleted} 条, 操作日志 {operation_deleted} 条")

    def cleanup_temp_files(self):
        """清理临时文件"""
        import os
        import glob

        temp_patterns = [
            "C:\\InventoryPro\\data\\temp\\*",
            "C:\\InventoryPro\\data\\export\\tmp_*",
            "C:\\InventoryPro\\data\\logs\\app_*.log.bak"
        ]

        deleted_count = 0
        deleted_size = 0

        for pattern in temp_patterns:
            for file_path in glob.glob(pattern):
                try:
                    file_size = os.path.getsize(file_path)
                    os.remove(file_path)
                    deleted_count += 1
                    deleted_size += file_size
                except Exception as e:
                    print(f"删除文件失败 {file_path}: {str(e)}")

        print(f"清理临时文件: {deleted_count} 个, 释放空间: {deleted_size / 1024 / 1024:.2f} MB")

    def optimize_database(self):
        """优化数据库"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        # 分析表统计信息
        cursor.execute("ANALYZE")

        # 重建索引
        cursor.execute("REINDEX")

        # 清理碎片
        cursor.execute("VACUUM")

        conn.commit()
        conn.close()

        print("数据库优化完成")

# 执行清理
if __name__ == "__main__":
    cleanup = DataCleanup()
    cleanup.cleanup_old_logs(days=30)
    cleanup.cleanup_temp_files()
    cleanup.optimize_database()
```

```python
# data_archive.py
import sqlite3
import shutil
from datetime import datetime, timedelta
import json

class DataArchiver:
    """数据归档工具"""

    def __init__(self):
        self.db_path = "C:\\InventoryPro\\data\\inventory.db"
        self.archive_path = "C:\\InventoryPro\\data\\archive"

    def archive_old_data(self, months=12):
        """归档旧数据"""
        cutoff_date = datetime.now() - timedelta(months=months)
        archive_filename = f"archive_{datetime.now().strftime('%Y%m%d_%H%M%S')}.db"
        archive_db_path = os.path.join(self.archive_path, archive_filename)

        # 创建归档目录
        os.makedirs(self.archive_path, exist_ok=True)

        # 复制数据库
        shutil.copy2(self.db_path, archive_db_path)

        # 连接归档数据库
        archive_conn = sqlite3.connect(archive_db_path)
        archive_cursor = archive_conn.cursor()

        # 创建归档记录
        archive_cursor.execute("""
            CREATE TABLE IF NOT EXISTS archive_info (
                id INTEGER PRIMARY KEY,
                archive_date DATETIME,
                cutoff_date DATETIME,
                record_counts TEXT,
                status TEXT
            )
        """)

        # 获取归档前的记录数
        record_counts = self._get_record_counts(archive_db_path)

        # 删除旧数据（保留结构）
        self._delete_old_data(archive_conn, cutoff_date)

        # 记录归档信息
        archive_cursor.execute("""
            INSERT INTO archive_info (
                archive_date, cutoff_date, record_counts, status
            ) VALUES (?, ?, ?, 'COMPLETED')
        """, (
            datetime.now().isoformat(),
            cutoff_date.isoformat(),
            json.dumps(record_counts)
        ))

        archive_conn.commit()
        archive_conn.close()

        # 优化主数据库
        self._optimize_main_database()

        print(f"数据归档完成: {archive_filename}")
        print(f"归档记录数: {record_counts}")

    def restore_from_archive(self, archive_db_path, target_date=None):
        """从归档恢复数据"""
        if not os.path.exists(archive_db_path):
            raise FileNotFoundError(f"归档文件不存在: {archive_db_path}")

        # 连接归档数据库
        archive_conn = sqlite3.connect(archive_db_path)

        # 查询归档信息
        cursor = archive_conn.cursor()
        cursor.execute("""
            SELECT * FROM archive_info
            WHERE status = 'COMPLETED'
            ORDER BY archive_date DESC
        """)

        archives = cursor.fetchall()

        if not archives:
            raise ValueError("没有可用的归档记录")

        # 选择归档记录
        if target_date:
            # 恢复指定日期的归档
            archive_info = next(
                (a for a in archives if a[2] <= target_date),
                None
            )
        else:
            # 恢复最新的归档
            archive_info = archives[0]

        if not archive_info:
            raise ValueError("找不到符合条件的归档记录")

        # 执行恢复
        self._perform_restore(archive_db_path, archive_info)

        print("数据恢复完成")
```

```python
# fix_db_lock.py
import sqlite3
import time
import psutil

def fix_database_lock():
    """修复数据库锁定问题"""

    print("检查数据库锁定状态...")

    # 查找占用数据库的进程
    db_path = "C:\\InventoryPro\\data\\inventory.db"
    lock_file = db_path + "-wal"

    if os.path.exists(lock_file):
        print("发现WAL锁文件，检查相关进程...")

        # 查找使用数据库的进程
        for proc in psutil.process_iter(['pid', 'name', 'open_files']):
            try:
                for file in proc.info['open_files'] or []:
                    if db_path in file.path:
                        print(f"进程 {proc.info['name']} (PID: {proc.info['pid']}) 正在使用数据库")

                        # 尝试优雅关闭
                        try:
                            proc.terminate()
                            proc.wait(timeout=5)
                            print(f"已终止进程 {proc.info['pid']}")
                        except psutil.TimeoutExpired:
                            print(f"强制终止进程 {proc.info['pid']}")
                            proc.kill()
            except (psutil.NoSuchProcess, psutil.AccessDenied):
                continue

        # 删除锁文件
        time.sleep(2)
        try:
            os.remove(lock_file)
            print("已删除锁文件")
        except Exception as e:
            print(f"删除锁文件失败: {str(e)}")

    # 检查数据库完整性
    conn = sqlite3.connect(db_path)
    try:
        cursor = conn.cursor()
        cursor.execute("PRAGMA integrity_check")
        result = cursor.fetchone()

        if result[0] == "ok":
            print("数据库完整性检查通过")
        else:
            print(f"数据库损坏: {result}")
            print("尝试修复...")
            cursor.execute("PRAGMA quick_check")
            quick_check = cursor.fetchone()

            if quick_check[0] == "ok":
                print("快速检查通过")
            else:
                print("需要从备份恢复")
                return False

    finally:
        conn.close()

    return True

if __name__ == "__main__":
    if fix_database_lock():
        print("数据库锁定问题已解决")
    else:
        print("需要从备份恢复数据库")
```

```python
def diagnose_performance():

    # 查看常用查询模式
    common_queries = [
        ("inbound_orders", "order_date"),
        ("outbound_orders", "order_date"),
        ("inventory", "product_id"),
        ("products", "code"),
        ("products", "name"),
    ]

    for table, column in common_queries:
        cursor.execute(f"""
            SELECT name FROM sqlite_master
            WHERE type = 'index' AND tbl_name = '{table}'
            AND sql LIKE '%{column}%'
        """)

        indexes = cursor.fetchall()
        if not indexes:
            print(f"警告: {table}.{column} 缺少索引")

    # 检查数据库统计信息
    print("\n3. 检查数据库统计信息...")
    cursor.execute("PRAGMA table_count")
    table_count = cursor.fetchone()[0]
    print(f"表数量: {table_count}")

    cursor.execute("PRAGMA index_count")
    index_count = cursor.fetchone()[0]
    print(f"索引数量: {index_count}")

    # 检查数据库大小
    cursor.execute("PRAGMA page_count")
    page_count = cursor.fetchone()[0]

    cursor.execute("PRAGMA page_size")
    page_size = cursor.fetchone()[0]

    db_size = page_count * page_size / 1024 / 1024
    print(f"数据库大小: {db_size:.2f} MB")

    conn.close()

    print("\n诊断完成，请查看以上输出")
```

```python
# performance_optimization.py

class PerformanceOptimizer:
    """性能优化器"""

    def __init__(self):
        self.config = self.load_config()

    def optimize_query_cache(self):
        """优化查询缓存"""

        # 实现查询结果缓存
        cache = {}

        def cached_query(query, params, ttl=300):
            """带缓存的查询"""
            cache_key = (query, tuple(params))

            if cache_key in cache:
                cached_result, timestamp = cache[cache_key]
                if time.time() - timestamp < ttl:
                    return cached_result

            # 执行查询
            result = execute_query(query, params)

            # 缓存结果
            cache[cache_key] = (result, time.time())

            return result

        return cached_query

    def implement_connection_pool(self):
        """实现连接池"""

        import queue
        import threading

        class ConnectionPool:
            def __init__(self, max_connections=5):
                self.pool = queue.Queue(maxsize=max_connections)
                self.lock = threading.Lock()

                # 创建初始连接
                for _ in range(max_connections):
                    conn = sqlite3.connect(
                        DB_FILE,
                        check_same_thread=False,
                        timeout=30
                    )
                    conn.execute("PRAGMA journal_mode = WAL")
                    conn.execute("PRAGMA synchronous = NORMAL")
                    self.pool.put(conn)

            def get_connection(self):
                return self.pool.get()

            def return_connection(self, conn):
                self.pool.put(conn)
```

### 8.2 日志分析脚本

```python
# log_analyzer.py
import re
from collections import defaultdict, Counter
from datetime import datetime

class LogAnalyzer:
    """日志分析器"""

    def __init__(self, log_file):
        self.log_file = log_file
        self.patterns = {
            'login': r'(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}).*?用户登录.*?用户名: (.*?), 状态: (成功|失败)',
            'error': r'(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}).*?ERROR.*?-(.*)',
            'operation': r'(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}).*?数据操作.*?用户: (.*?), 操作: (.*?), 表: (.*?)'
        }

    def analyze_login_logs(self, start_date=None, end_date=None):
        """分析登录日志"""
        login_stats = {
            'total': 0,
            'success': 0,
            'failed': 0,
            'users': Counter(),
            'failed_attempts': defaultdict(list)
        }

        with open(self.log_file, 'r', encoding='utf-8') as f:
            for line in f:
                match = re.search(self.patterns['login'], line)
                if match:
                    timestamp, username, status = match.groups()

                    # 时间过滤
                    log_time = datetime.strptime(timestamp, '%Y-%m-%d %H:%M:%S')
                    if start_date and log_time < start_date:
                        continue
                    if end_date and log_time > end_date:
                        continue

                    login_stats['total'] += 1
                    login_stats['users'][username] += 1

                    if status == '成功':
                        login_stats['success'] += 1
                    else:
                        login_stats['failed'] += 1
                        login_stats['failed_attempts'][username].append(log_time)

        return login_stats

    def analyze_errors(self, hours=24):
        """分析错误日志"""
        error_stats = defaultdict(int)
        recent_errors = []
        cutoff_time = datetime.now() - timedelta(hours=hours)

        with open(self.log_file, 'r', encoding='utf-8') as f:
            for line in f:
                match = re.search(self.patterns['error'], line)
                if match:
                    timestamp, error_msg = match.groups()
                    log_time = datetime.strptime(timestamp, '%Y-%m-%d %H:%M:%S')

                    if log_time >= cutoff_time:
                        error_stats[error_msg] += 1
                        recent_errors.append({
                            'timestamp': timestamp,
                            'message': error_msg
                        })

        return {
            'error_counts': error_stats,
            'recent_errors': recent_errors[:100]  # 最近100条错误
        }

    def generate_report(self):
        """生成分析报告"""
        print("=" * 60)
        print("进销存系统日志分析报告")
        print("=" * 60)

        # 登录分析
        print("\n1. 登录统计（最近24小时）")
        login_stats = self.analyze_login_logs(
            start_date=datetime.now() - timedelta(hours=24)
        )

        if login_stats['total'] > 0:
            print(f"总登录次数: {login_stats['total']}")
            print(f"成功登录: {login_stats['success']}")
            print(f"失败登录: {login_stats['failed']}")
            print(f"成功率: {login_stats['success']/login_stats['total']*100:.2f}%")

            print("\n最活跃用户:")
            for user, count in login_stats['users'].most_common(5):
                print(f"  {user}: {count}次")

            if login_stats['failed_attempts']:
                print("\n异常登录:")
                for user, attempts in login_stats['failed_attempts'].items():
                    if len(attempts) > 3:
                        print(f"  {user}: {len(attempts)}次失败")

        # 错误分析
        print("\n2. 错误统计（最近24小时）")
        error_stats = self.analyze_errors(hours=24)

        if error_stats['error_counts']:
            print("\n常见错误:")
            for error, count in sorted(
                error_stats['error_counts'].items(),
                key=lambda x: x[1],
                reverse=True
            )[:5]:
                print(f"  [{count}次] {error}")

        if error_stats['recent_errors']:
            print("\n最新错误:")
            for error in error_stats['recent_errors'][:5]:
                print(f"  {error['timestamp']} - {error['message']}")

if __name__ == "__main__":
    analyzer = LogAnalyzer("C:\\InventoryPro\\data\\logs\\app.log")
    analyzer.generate_report()
```
