# performance_profiler.py
import cProfile
import pstats
import time
import sqlite3
from functools import wraps

def profile_performance(func):
    """性能分析装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        pr = cProfile.Profile()
        pr.enable()

        start_time = time.time()
        result = func(*args, **kwargs)
        end_time = time.time()

        pr.disable()

        # 保存性能数据
        stats = pstats.Stats(pr)
        stats.sort_stats('cumulative')
        stats.print_stats(20)  # 显示前20个最耗时的函数

        print(f"\n函数 {func.__name__} 执行时间: {end_time - start_time:.2f}秒")

        return result
    return wrapper

class PerformanceMonitor:
    """性能监控器"""

    def __init__(self):
        self.metrics = {
            'query_times': [],
            'operation_counts': {},
            'memory_usage': [],
            'cpu_usage': []
        }

    def monitor_database_query(self, query: str, params: tuple = ()):
        """监控数据库查询性能"""
        start_time = time.time()

        conn = sqlite3.connect("C:\\InventoryPro\\data\\inventory.db")
        cursor = conn.cursor()

        # 执行查询前检查
        cursor.execute("EXPLAIN QUERY PLAN " + query, params)
        plan = cursor.fetchall()

        # 执行查询
        cursor.execute(query, params)
        result = cursor.fetchall()

        execution_time = time.time() - start_time
        self.metrics['query_times'].append({
            'query': query[:100] + '...' if len(query) > 100 else query,
            'time': execution_time,
            'rows': len(result),
            'plan': plan
        })

        conn.close()

        # 分析查询计划
        self.analyze_query_plan(query, plan)

        return result

    def analyze_query_plan(self, query, plan):
        """分析查询执行计划"""
        # 检查是否使用了索引
        using_index = any('USING INDEX' in str(row) for row in plan)

        if not using_index and any(keyword in query.upper() for keyword in ['WHERE', 'JOIN']):
            print(f"[警告] 查询未使用索引: {query[:50]}...")

        # 检查是否进行了全表扫描
        full_scan = any('SCAN TABLE' in str(row) for row in plan)
        if full_scan:
            print(f"[警告] 查询进行了全表扫描: {query[:50]}...")

import sqlite3

class IndexAnalyzer:
    """索引分析器"""

    def __init__(self, db_path):
        self.db_path = db_path

    def analyze_index_usage(self):
        """分析索引使用情况"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        print("\n=== 索引使用情况分析 ===\n")

        # 1. 检查所有索引
        cursor.execute("""
            SELECT name, tbl_name, sql
            FROM sqlite_master
            WHERE type = 'index'
            AND name NOT LIKE 'sqlite_%'
            ORDER BY tbl_name, name
        """)

        indexes = cursor.fetchall()

        # 按表分组
        tables = {}
        for idx in indexes:
            table = idx[1]
            if table not in tables:
                tables[table] = []
            tables[table].append({
                'name': idx[0],
                'sql': idx[2] if idx[2] else '自动生成'
            })

        # 显示索引信息
        for table, idx_list in tables.items():
            print(f"表: {table}")
            for idx in idx_list:
                print(f"  索引: {idx['name']}")
                print(f"    定义: {idx['sql']}")
            print()

        # 2. 检查未使用的索引
        print("\n=== 检查未使用的索引 ===\n")
        cursor.execute("PRAGMA index_list('products')")
        product_indexes = cursor.fetchall()

        print("产品表索引:")
        for idx in product_indexes:
            idx_name = idx[1]
            is_unique = idx[2]
            origin = idx[3]
            partial = idx[4]

            print(f"  {idx_name}")
            print(f"    唯一: {is_unique}")
            print(f"    来源: {origin}")
            print(f"    部分索引: {partial}")

            # 检查索引列
            cursor.execute(f"PRAGMA index_info('{idx_name}')")
            columns = cursor.fetchall()
            print(f"    列: {', '.join([col[2] for col in columns])}")

        conn.close()

import sqlite3
import time

class QueryOptimizer:
    """查询优化器"""

    def __init__(self, db_path):
        self.db_path = db_path
        self.query_cache = {}

    def execute_optimized_query(self, query: str, params: tuple = ()):
        """执行优化的查询"""
        # 检查缓存
        cache_key = (query, params)
        if cache_key in self.query_cache:
            cached_result, timestamp = self.query_cache[cache_key]
            # 缓存有效期5分钟
            if time.time() - timestamp < 300:
                return cached_result

        # 优化查询
        optimized_query = self.optimize_query(query)

        # 执行查询
        conn = sqlite3.connect(self.db_path)
        conn.row_factory = sqlite3.Row
        cursor = conn.cursor()

        start_time = time.time()
        cursor.execute(optimized_query, params)
        result = cursor.fetchall()
        execution_time = time.time() - start_time

        conn.close()

        # 缓存结果（仅对SELECT查询）
        if query.strip().upper().startswith('SELECT'):
            self.query_cache[cache_key] = (result, time.time())

        # 记录慢查询
        if execution_time > 1.0:  # 超过1秒
            self.log_slow_query(query, execution_time)

        return [dict(row) for row in result]

                p.category,
                p.unit,

                i.location,

                i.last_updated,

                    WHEN i.quantity <= i.min_stock THEN '库存不足'
                    WHEN i.quantity <= i.min_stock * 1.2 THEN '库存偏低'
                    ELSE '库存正常'
                END as stock_status
            FROM inventory i
            JOIN products p ON i.product_id = p.id
            LEFT JOIN warehouses w ON i.warehouse_id = w.id
            WHERE p.active = 1
        """

        # 动态添加过滤条件
        if filters:
            if filters.get('category'):
                base_query += " AND p.category = ?"
                params.append(filters['category'])

            if filters.get('warehouse'):
                base_query += " AND i.warehouse_id = ?"
                params.append(filters['warehouse'])

            if filters.get('location'):
                base_query += " AND i.location LIKE ?"
                params.append(f"%{filters['location']}%")

            if filters.get('stock_status'):
                if filters['stock_status'] == 'low':

                elif filters['stock_status'] == 'normal':
                    base_query += " AND i.quantity > i.min_stock * 1.2"

        # 添加排序和分页
        base_query += " ORDER BY p.code"
        base_query += " LIMIT ? OFFSET ?"
        params.extend([100, 0])  # 默认每页100条

        return self.execute_optimized_query(base_query, tuple(params))

                SELECT
                    ds.sale_date,
                    p.code,
                    p.name,
                    p.category,
                    ds.daily_quantity,
                    ds.daily_amount,
                    p.purchase_price,
                    (ds.daily_amount - (ds.daily_quantity * p.purchase_price)) as profit
                FROM daily_sales ds
                JOIN products p ON ds.product_id = p.id
                ORDER BY ds.sale_date, ds.daily_amount DESC

        return self.execute_optimized_query(query, (start_date, end_date))

import sqlite3

class DatabaseConfig:
    """数据库配置优化"""

    @staticmethod
    def optimize_connection(db_path: str):
        """优化数据库连接"""
        conn = sqlite3.connect(db_path)

        # 执行优化设置
        settings = [
            "PRAGMA journal_mode = WAL",
            "PRAGMA synchronous = NORMAL",
            "PRAGMA cache_size = 10000",
            "PRAGMA temp_store = MEMORY",
            "PRAGMA mmap_size = 268435456",
            "PRAGMA busy_timeout = 30000",
            "PRAGMA foreign_keys = ON",
            "PRAGMA optimize"
        ]

        for setting in settings:
            conn.execute(setting)

        return conn

    @staticmethod
    def create_connection_pool(max_connections: int = 5):
        """创建数据库连接池"""
        import queue
        import threading

        class ConnectionPool:
            def __init__(self, db_path, max_connections):
                self.db_path = db_path
                self.pool = queue.Queue(maxsize=max_connections)
                self.lock = threading.Lock()

                # 预创建连接
                for _ in range(max_connections):
                    conn = DatabaseConfig.optimize_connection(db_path)
                    self.pool.put(conn)

            def get_connection(self):
                return self.pool.get()

            def return_connection(self, conn):
                # 重置连接状态
                conn.execute("PRAGMA optimize")
                self.pool.put(conn)

            def close_all(self):
                while not self.pool.empty():
                    conn = self.pool.get()
                    conn.close()

        return ConnectionPool("C:\\InventoryPro\\data\\inventory.db", max_connections)

import threading
from concurrent.futures import ThreadPoolExecutor
import queue

class DataLoader:
    """数据加载器（支持异步和分页）"""

    def __init__(self, db_path):
        self.db_path = db_path
        self.executor = ThreadPoolExecutor(max_workers=4)
        self.cache = {}

    def load_data_async(self, query: str, params: tuple, callback):
        """异步加载数据"""
        def worker():
            try:
                # 执行查询
                conn = sqlite3.connect(self.db_path)
                cursor = conn.cursor()
                cursor.execute(query, params)
                data = cursor.fetchall()
                conn.close()

                # 在主线程执行回调
                import tkinter as tk
                if hasattr(callback, '__self__') and isinstance(callback.__self__, tk.Misc):
                    callback.__self__.after(0, lambda: callback(data))
                else:
                    callback(data)
            except Exception as e:
                print(f"数据加载失败: {str(e)}")

        self.executor.submit(worker)

    def load_paginated_data(self, table: str, page: int, page_size: int = 100, filters: dict = None):
        """分页加载数据"""
        offset = (page - 1) * page_size

        # 构建查询
        query = f"SELECT * FROM {table}"
        params = []

        if filters:
            where_clauses = []
            for key, value in filters.items():
                where_clauses.append(f"{key} LIKE ?")
                params.append(f"%{value}%")
            query += " WHERE " + " AND ".join(where_clauses)

        query += " ORDER BY id DESC LIMIT ? OFFSET ?"
        params.extend([page_size, offset])

        # 获取总数
        count_query = f"SELECT COUNT(*) FROM {table}"
        if filters:
            query_count = count_query + " WHERE " + " AND ".join(where_clauses)
        else:
            query_count = count_query

        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        # 获取总数
        cursor.execute(query_count, params[:-2])
        total_count = cursor.fetchone()[0]

        # 获取数据
        cursor.execute(query, params)
        data = cursor.fetchall()

        conn.close()

        return {
            'data': data,
            'page': page,
            'page_size': page_size,
            'total_count': total_count,
            'total_pages': (total_count + page_size - 1) // page_size
        }

    def preload_related_data(self, product_ids):
        """预加载相关数据"""
        if not product_ids:
            return

        # 构建IN查询
        placeholders = ','.join(['?'] * len(product_ids))

        # 预加载库存
        inventory_query = f"""
            SELECT product_id, SUM(quantity) as total_quantity
            FROM inventory
            WHERE product_id IN ({placeholders})
            GROUP BY product_id
        """

        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute(inventory_query, product_ids)
        inventory_data = {row[0]: row[1] for row in cursor.fetchall()}
        conn.close()

        # 缓存数据
        self.cache['inventory'] = inventory_data

        return inventory_data

            visible_count = tree_height // item_height + 5

            start = max(0, self.visible_start - 10)

            # 清除不在可见范围的项目

                    self.tree.delete(item)

            for i in range(start, end):
                if i >= len(self.tree.get_children()):
                    data = self.data_source.get_item(i)
                    if data:
                        self.tree.insert('', 'end', values=data)

import json
import hashlib
from typing import Any, Optional

class CacheManager:
    """缓存管理器"""

    def __init__(self, max_size: int = 1000, ttl: int = 300):
        self.cache = {}
        self.access_times = {}
        self.max_size = max_size
        self.ttl = ttl  # 生存时间（秒）

    def get(self, key: str) -> Optional[Any]:
        """获取缓存"""
        if key in self.cache:
            data, timestamp = self.cache[key]

            # 检查是否过期
            if time.time() - timestamp < self.ttl:
                self.access_times[key] = time.time()
                return data
            else:
                # 过期，删除
                del self.cache[key]
                if key in self.access_times:
                    del self.access_times[key]

        return None

    def set(self, key: str, value: Any):
        """设置缓存"""
        # 如果缓存已满，删除最久未访问的项
        if len(self.cache) >= self.max_size:
            oldest_key = min(self.access_times.keys(), key=lambda k: self.access_times[k])
            del self.cache[oldest_key]
            del self.access_times[oldest_key]

        self.cache[key] = (value, time.time())
        self.access_times[key] = time.time()

    def delete(self, key: str):
        """删除缓存"""
        if key in self.cache:
            del self.cache[key]
        if key in self.access_times:
            del self.access_times[key]

    def clear(self):
        """清空缓存"""
        self.cache.clear()
        self.access_times.clear()

    def generate_key(self, query: str, params: tuple = ()) -> str:
        """生成缓存键"""
        key_data = query + str(params)
        return hashlib.md5(key_data.encode()).hexdigest()

class AsyncTaskManager:
    """异步任务管理器"""

    def __init__(self, max_workers: int = 4):
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
        self.task_queue = queue.Queue()
        self.results = {}
        self.running = False
        self.progress_callbacks = {}

    def submit_task(self, task_id: str, func, *args, **kwargs):
        """提交异步任务"""
        future = self.executor.submit(self._run_task, task_id, func, *args, **kwargs)
        self.results[task_id] = {
            'future': future,
            'status': 'pending',
            'progress': 0,
            'result': None,
            'error': None
        }
        return task_id

    def _run_task(self, task_id: str, func, *args, **kwargs):
        """运行任务"""
        try:
            self.results[task_id]['status'] = 'running'

            # 如果函数支持进度回调，添加回调
            if 'progress_callback' in func.__code__.co_varnames:
                kwargs['progress_callback'] = lambda p: self.update_progress(task_id, p)

            # 执行函数
            result = func(*args, **kwargs)
            self.results[task_id]['result'] = result
            self.results[task_id]['status'] = 'completed'

            return result
        except Exception as e:
            self.results[task_id]['error'] = str(e)
            self.results[task_id]['status'] = 'failed'
            raise

    def update_progress(self, task_id: str, progress: int):
        """更新任务进度"""
        if task_id in self.results:
            self.results[task_id]['progress'] = progress

            # 调用进度回调
            if task_id in self.progress_callbacks:
                callback = self.progress_callbacks[task_id]
                callback(progress)

    def get_task_status(self, task_id: str):
        """获取任务状态"""
        return self.results.get(task_id, {})

    def set_progress_callback(self, task_id: str, callback):
        """设置进度回调"""

class BatchDataProcessor:
    """批量数据处理器"""

    def __init__(self, db_path, batch_size: int = 1000):
        self.db_path = db_path
        self.batch_size = batch_size

    def batch_insert(self, table: str, data_list: list, progress_callback=None):
        """批量插入数据"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        total = len(data_list)
        processed = 0

        try:
            # 开始事务
            conn.execute("BEGIN TRANSACTION")

            # 分批处理
            for i in range(0, total, self.batch_size):
                batch = data_list[i:i + self.batch_size]

                # 构建批量插入SQL
                if batch:
                    placeholders = ','.join(['?' for _ in range(len(batch[0]))])
                    sql = f"INSERT INTO {table} VALUES ({placeholders})"
                    cursor.executemany(sql, batch)

                    processed += len(batch)

                    # 更新进度
                    if progress_callback:
                        progress = int((processed / total) * 100)
                        progress_callback(progress)

                    # 提交批次
                    conn.commit()
                    conn.execute("BEGIN TRANSACTION")

            # 完成事务
            conn.commit()

        except Exception as e:
            conn.rollback()
            raise e
        finally:
            conn.close()

        return processed

    def batch_update(self, table: str, updates: list, progress_callback=None):
        """批量更新数据"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        total = len(updates)
        processed = 0

        try:
            conn.execute("BEGIN TRANSACTION")

            for update in updates:
                # update格式: {'id': 1, 'field1': 'value1', 'field2': 'value2'}
                set_clause = ', '.join([f"{k} = ?" for k in update.keys() if k != 'id'])
                values = [v for k, v in update.items() if k != 'id']
                values.append(update['id'])

                sql = f"UPDATE {table} SET {set_clause} WHERE id = ?"
                cursor.execute(sql, values)

                processed += 1

                if progress_callback and processed % 100 == 0:
                    progress = int((processed / total) * 100)
                    progress_callback(progress)

            conn.commit()

        except Exception as e:
            conn.rollback()
            raise e
        finally:
            conn.close()

        return processed

class FileStreamer:
    """文件流处理器"""

    def __init__(self, chunk_size: int = 8192):
        self.chunk_size = chunk_size

    def stream_read(self, file_path: str):
        """流式读取大文件"""
        with open(file_path, 'rb') as f:
            while True:
                chunk = f.read(self.chunk_size)
                if not chunk:
                    break
                yield chunk

    def stream_write(self, file_path: str, data_stream):
        """流式写入大文件"""
        with open(file_path, 'wb') as f:
            for chunk in data_stream:
                f.write(chunk)

    def stream_process(self, input_path: str, output_path: str, processor):
        """流式处理文件"""
        with open(input_path, 'rb') as infile, \
             open(output_path, 'wb') as outfile:

            while True:
                chunk = infile.read(self.chunk_size)
                if not chunk:
                    break

                processed_chunk = processor(chunk)
                if processed_chunk:
                    outfile.write(processed_chunk)

                    results.append(result)

import tkinter as tk

class OptimizedReportGenerator:
    """优化的报表生成器"""

    def __init__(self):
        self.progress_queue = queue.Queue()
        self.data_queue = queue.Queue()

    def generate_sales_report_async(self, start_date, end_date, callback):
        """异步生成销售报表"""
        def worker():
            try:
                # 1. 准备数据（分批查询）
                total_steps = 5
                current_step = 0

                # 步骤1: 获取订单总数
                self.progress_queue.put(('progress', 0, '获取订单总数...'))
                order_count = self.get_order_count(start_date, end_date)
                current_step += 1

                # 步骤2: 分批处理订单
                batch_size = 1000
                processed = 0
                report_data = []

                while processed < order_count:
                    # 查询一批数据
                    batch_data = self.get_order_batch(
                        start_date, end_date, processed, batch_size
                    )

                    # 处理数据
                    for order in batch_data:
                        # 计算利润等
                        order_data = self.process_order(order)
                        if order_data:
                            report_data.append(order_data)
                            # 发送到队列供UI更新
                            if len(report_data) % 100 == 0:
                                self.data_queue.put(('data', report_data[-100:]))

                    processed += len(batch_data)
                    progress = (current_step + processed / order_count) / total_steps * 100
                    self.progress_queue.put(('progress', progress, f'处理中... {processed}/{order_count}'))

                # 步骤3: 计算汇总
                self.progress_queue.put(('progress', 80, '计算汇总...'))
                summary = self.calculate_summary(report_data)

                # 步骤4: 生成图表数据
                self.progress_queue.put(('progress', 90, '生成图表...'))
                charts_data = self.generate_charts_data(report_data)

                # 步骤5: 完成
                self.progress_queue.put(('complete', 100, '报表生成完成'))
                self.data_queue.put(('result', {
                    'data': report_data,
                    'summary': summary,
                    'charts': charts_data
                }))

            except Exception as e:
                self.progress_queue.put(('error', 0, str(e)))

        thread = threading.Thread(target=worker, daemon=True)
        thread.start()
        return thread

    def get_order_count(self, start_date, end_date):

            WHERE o.order_date BETWEEN ? AND ?
            AND o.status = 'completed'
            ORDER BY o.order_date
            LIMIT ? OFFSET ?
        """, (start_date, end_date, limit, offset))

        results = cursor.fetchall()
        conn.close()
        return results

"""
库存查询优化案例

优化前：
- 查询10万条库存记录需要8-12秒
- 内存占用超过500MB
- UI响应缓慢

优化后：

- 内存占用稳定在200MB以下
- 支持实时过滤和排序
"""

class OptimizedInventoryQuery:
    """优化的库存查询"""

    def __init__(self):
        self.cache = {}
        self.index_cache = {}

    def fast_inventory_search(self, filters=None, sort_by='code', limit=1000):
        """快速库存搜索"""
        # 生成缓存键
        cache_key = self.generate_cache_key(filters, sort_by, limit)

        # 检查缓存
        if cache_key in self.cache:
            print("从缓存获取数据")
            return self.cache[cache_key]

        # 构建优化的SQL查询
        query = self.build_optimized_query(filters, sort_by, limit)

        # 执行查询
        start_time = time.time()
        results = self.execute_query(query)
        query_time = time.time() - start_time

        print(f"查询耗时: {query_time:.2f}秒")

        # 缓存结果
        self.cache[cache_key] = results

        return results

    def build_optimized_query(self, filters, sort_by, limit):
        """构建优化的查询"""
        # 使用索引列作为排序优先级
        sort_map = {
            'code': 'p.code',
            'name': 'p.name',
            'quantity': 'i.quantity',
            'location': 'i.location'
        }

        base_query = """
            SELECT
                p.id,
                p.code,
                p.name,
                p.category,
                p.unit,
                i.quantity,
                i.location,
                i.min_stock,
                w.name as warehouse
            FROM inventory i
            JOIN products p ON i.product_id = p.id
            LEFT JOIN warehouses w ON i.warehouse_id = w.id
            WHERE 1=1
        """

        params = []

        # 添加过滤条件
        if filters:
            if filters.get('category'):
                base_query += " AND p.category = ?"
                params.append(filters['category'])

            if filters.get('low_stock'):
                base_query += " AND i.quantity <= i.min_stock"
                params.append(filters['low_stock'])

            if filters.get('search'):
                base_query += " AND (p.code LIKE ? OR p.name LIKE ?)"
                search_term = f"%{filters['search']}%"
                params.extend([search_term, search_term])

        # 添加排序
        if sort_by in sort_map:
            base_query += f" ORDER BY {sort_map[sort_by]}"

        # 添加限制
        if limit:
            base_query += " LIMIT ?"
            params.append(limit)

        return base_query, params

    def execute_query(self, query_params):
        """执行查询"""
        query, params = query_params

        conn = sqlite3.connect("C:\\InventoryPro\\data\\inventory.db")
        conn.row_factory = sqlite3.Row
        cursor = conn.cursor()

        # 使用预编译语句
        cursor.execute(query, params)
        results = cursor.fetchall()

        conn.close()

        return [dict(row) for row in results]
